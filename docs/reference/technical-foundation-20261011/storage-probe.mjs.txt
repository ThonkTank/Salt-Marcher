import Database from 'better-sqlite3';
import {mkdirSync,writeFileSync,readFileSync,openSync,closeSync,fsyncSync,renameSync,statSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {performance} from 'node:perf_hooks';
import assert from 'node:assert/strict';
const root=new URL('./probe-data/',import.meta.url).pathname;
mkdirSync(root,{recursive:true});
const filename=root+'store.sqlite';
function open(){const d=new Database(filename);d.pragma('journal_mode=WAL');d.pragma('synchronous=FULL');d.pragma('foreign_keys=ON');return d;}
if(process.argv[2]==='fault'){
 const d=open(),point=process.argv[3];
 if(point==='before')process.kill(process.pid,'SIGKILL');
 d.exec('BEGIN IMMEDIATE');
 d.prepare('UPDATE record SET revision=revision+1, body=? WHERE id IN (1,2,3)').run('{"geometry":"new","key":"new","actor":"new"}');
 d.prepare('INSERT INTO receipt VALUES (?,?)').run('fault-'+point,'applied');
 if(point==='during')process.kill(process.pid,'SIGKILL');
 d.exec('COMMIT');
 process.kill(process.pid,'SIGKILL');
}
let d=open();
d.exec('DROP TABLE IF EXISTS record; DROP TABLE IF EXISTS receipt; CREATE TABLE record(id INTEGER PRIMARY KEY, revision INTEGER NOT NULL, body TEXT NOT NULL) STRICT; CREATE TABLE receipt(id TEXT PRIMARY KEY, body TEXT NOT NULL) STRICT;');
const body=JSON.stringify({name:'Unchanged authored text',geometry:[[1,2],[3,4],[5,6]],notes:'Author words and links [[room-id]] '.repeat(8)});
const records=Array.from({length:100000},(_,i)=>({id:i+1,revision:0,body}));
const seed=d.prepare('INSERT INTO record VALUES(?,?,?)');d.transaction(()=>{for(const r of records)seed.run(r.id,r.revision,r.body)})();
d.pragma('wal_checkpoint(TRUNCATE)');
const percentile=a=>[...a].sort((x,y)=>x-y)[Math.ceil(.95*a.length)-1];
const sqlite=[],json=[],queries=[];
const update=d.prepare('UPDATE record SET revision=revision+1, body=? WHERE id=? AND revision=?');
const tx=d.transaction((i)=>{assert.equal(update.run(body,50000,i).changes,1);d.prepare('INSERT INTO receipt VALUES (?,?)').run('edit-'+i,JSON.stringify({before:i,after:i+1,undo:body}));});
for(let i=0;i<30;i++){
 let t=performance.now();tx(i);sqlite.push(performance.now()-t);
 records[49999].revision=i+1;t=performance.now();const serialized=JSON.stringify(records);
 const fd=openSync(root+'aggregate.pending','w');writeFileSync(fd,serialized);fsyncSync(fd);closeSync(fd);renameSync(root+'aggregate.pending',root+'aggregate.json');
 const directory=openSync(root,'r');fsyncSync(directory);closeSync(directory);json.push(performance.now()-t);
 t=performance.now();d.prepare('SELECT id,revision,body FROM record WHERE id BETWEEN ? AND ?').all(49872,50127);queries.push(performance.now()-t);
}
assert.equal(update.run(body,50000,0).changes,0);
assert.equal(d.prepare('SELECT count(*) n FROM receipt').get().n,30);
d.close();
const crash=[];
for(const point of ['before','during','after']){
 const child=spawnSync(process.execPath,[new URL(import.meta.url).pathname,'fault',point],{encoding:'utf8'});
 assert.equal(child.signal,'SIGKILL');d=open();
 const rows=d.prepare('SELECT revision FROM record WHERE id IN (1,2,3)').all();
 assert.ok(rows.every(r=>r.revision===(point==='after'?1:0)));
 assert.equal(d.prepare('SELECT count(*) n FROM receipt WHERE id=?').get('fault-'+point).n,point==='after'?1:0);
 assert.equal(d.pragma('integrity_check',{simple:true}),'ok');crash.push({point,rows:rows.map(r=>r.revision),receipt:point==='after'});d.close();
}
d=open();const bt=performance.now();await d.backup(root+'backup.sqlite');const backupMs=performance.now()-bt;d.close();
const backup=new Database(root+'backup.sqlite',{readonly:true});assert.equal(backup.prepare('SELECT count(*) n FROM record').get().n,100000);assert.equal(backup.pragma('integrity_check',{simple:true}),'ok');backup.close();
const report={date:new Date().toISOString(),node:process.version,sqlite:new Database(':memory:').prepare('select sqlite_version() v').get().v,recordCount:records.length,serializedBytes:statSync(root+'aggregate.json').size,sqliteBytes:statSync(filename).size,iterations:30,sqliteCommitP95Ms:percentile(sqlite),aggregateJSONCommitP95Ms:percentile(json),query256P95Ms:percentile(queries),backupMs,crash,staleRevisionRejected:true,scope:'isolated feasibility; not renderer, power-loss, production, or RP-R/RP-L acceptance',raw:{sqlite,json,queries}};
writeFileSync(root+'result.json',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify({...report,raw:undefined},null,2));
