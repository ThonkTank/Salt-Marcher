import Database from 'better-sqlite3';
import {mkdirSync,readFileSync,writeFileSync,copyFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import assert from 'node:assert/strict';
const root=new URL('./payload-probe-data/',import.meta.url).pathname;
mkdirSync(root,{recursive:true});
const file=root+'active.sqlite',backupFile=root+'backup.sqlite';
const hash=b=>createHash('sha256').update(b).digest('hex');
let d=new Database(file);d.pragma('journal_mode=WAL');d.pragma('synchronous=FULL');
d.exec('DROP TABLE IF EXISTS record; CREATE TABLE record(id TEXT PRIMARY KEY, revision INTEGER, body TEXT, checksum TEXT) STRICT;');
const body='{"notes":"PAYLOAD_SENTINEL_UNIQUE_A","geometry":"original"}';
d.prepare('INSERT INTO record VALUES(?,?,?,?)').run('damaged-unit',0,body,hash(body));
d.prepare('INSERT INTO record VALUES(?,?,?,?)').run('unaffected',0,'old',hash('old'));
await d.backup(backupFile);
d.prepare('UPDATE record SET revision=1,body=?,checksum=? WHERE id=?').run('newest confirmed',hash('newest confirmed'),'unaffected');
d.pragma('wal_checkpoint(TRUNCATE)');d.close();
const original=readFileSync(file),needle=Buffer.from('PAYLOAD_SENTINEL_UNIQUE_A'),offset=original.indexOf(needle);
assert.ok(offset>=0);const damaged=Buffer.from(original);damaged[offset+needle.length-1]^=1;writeFileSync(file,damaged);
copyFileSync(file,root+'original-damaged.sqlite');
d=new Database(file);const bad=d.prepare('SELECT * FROM record WHERE id=?').get('damaged-unit');
assert.notEqual(hash(bad.body),bad.checksum);
const survivor=d.prepare('SELECT * FROM record WHERE id=?').get('unaffected');assert.equal(survivor.body,'newest confirmed');assert.equal(hash(survivor.body),survivor.checksum);
const backup=new Database(backupFile,{readonly:true});const recovery=backup.prepare('SELECT * FROM record WHERE id=?').get('damaged-unit');assert.equal(hash(recovery.body),recovery.checksum);
d.transaction(()=>{d.prepare('UPDATE record SET revision=?,body=?,checksum=? WHERE id=?').run(recovery.revision,recovery.body,recovery.checksum,recovery.id)})();
assert.equal(d.prepare('SELECT body FROM record WHERE id=?').get('damaged-unit').body,body);assert.equal(d.prepare('SELECT body FROM record WHERE id=?').get('unaffected').body,'newest confirmed');
assert.equal(d.pragma('integrity_check',{simple:true}),'ok');d.close();backup.close();
assert.deepEqual(readFileSync(root+'original-damaged.sqlite'),damaged);
const report={date:new Date().toISOString(),fault:'one direct file-byte bit flip in authored JSON body; not SQLite metadata',checksumDetected:true,unaffectedLatestRecordRetained:true,backupUnitValidated:true,originalDamagedBytesRetained:true,recovery:'single independent unit restored from verified backup',limits:'not coupled-command closure, structural-page recovery, adversarial checksum replacement, or production acceptance'};
writeFileSync(root+'result.json',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report));
